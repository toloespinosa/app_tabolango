-- 1. Almacena los movimientos que entran/salen del banco
CREATE TABLE `app_movimientos_bancarios` (
  `id` int NOT NULL AUTO_INCREMENT,
  `fecha_movimiento` date NOT NULL,
  `descripcion` varchar(255) COLLATE utf8mb3_unicode_ci NOT NULL,
  `monto` decimal(12,2) NOT NULL,
  `tipo` enum('ingreso','egreso') COLLATE utf8mb3_unicode_ci DEFAULT 'ingreso',
  `estado` enum('pendiente','parcial','conciliado') COLLATE utf8mb3_unicode_ci DEFAULT 'pendiente',
  `saldo_disponible` decimal(12,2) NOT NULL COMMENT 'Permite pagar múltiples facturas con un solo abono',
  `banco` varchar(50) COLLATE utf8mb3_unicode_ci DEFAULT NULL,
  `fecha_registro` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
  PRIMARY KEY (`id`),
  INDEX `idx_estado` (`estado`),
  INDEX `idx_fecha` (`fecha_movimiento`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_unicode_ci;

-- 2. Almacena el historial de qué movimiento pagó qué pedido (El "Match")
CREATE TABLE `app_conciliaciones` (
  `id` int NOT NULL AUTO_INCREMENT,
  `id_movimiento` int NOT NULL,
  `id_pedido_interno` int NOT NULL COMMENT 'Se enlaza con pedidos_activos.id_interno',
  `monto_aplicado` decimal(12,2) NOT NULL,
  `fecha_conciliacion` timestamp NULL DEFAULT CURRENT_TIMESTAMP,
  `conciliado_por` varchar(100) COLLATE utf8mb3_unicode_ci NOT NULL,
  PRIMARY KEY (`id`),
  FOREIGN KEY (`id_movimiento`) REFERENCES `app_movimientos_bancarios`(`id`) ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb3 COLLATE=utf8mb3_unicode_ci;

-- 3. Preparamos la tabla de pedidos para saber si están pagados
ALTER TABLE `pedidos_activos`
ADD COLUMN `estado_pago` enum('Pendiente','Parcial','Pagado') COLLATE utf8mb3_unicode_ci DEFAULT 'Pendiente',
ADD COLUMN `monto_pagado` decimal(12,2) DEFAULT 0.00;

ALTER TABLE `app_movimientos_bancarios`
ADD COLUMN `fintoc_id` varchar(100) COLLATE utf8mb3_unicode_ci DEFAULT NULL AFTER `id`,
ADD UNIQUE INDEX `idx_fintoc_id` (`fintoc_id`);

-- 1. Borramos el historial de prueba viejo para bajarlo limpio
TRUNCATE TABLE `app_movimientos_bancarios`;

-- 2. Agregamos las columnas de RUT y Nombre

    OR color_diferenciador IS NULL
    OR color_diferenciador = '';

---------------- Módulo Cotizaciones de Compras / Proveedores (2026-05-27)
-- Tres tablas nuevas: proveedores, catálogo por proveedor, historial de precios.

CREATE TABLE proveedores (
    id_proveedor    INT AUTO_INCREMENT PRIMARY KEY,
    nombre          VARCHAR(150) NOT NULL,
    rut             VARCHAR(15)  DEFAULT NULL,
    contacto        VARCHAR(100) DEFAULT NULL,
    telefono        VARCHAR(20)  DEFAULT NULL,
    email           VARCHAR(100) DEFAULT NULL,
    ciudad          VARCHAR(100) DEFAULT NULL,
    notas           TEXT         DEFAULT NULL,
    activo          TINYINT(1)   DEFAULT 1,
    fecha_registro  TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_prov_activo (activo),
    INDEX idx_prov_nombre (nombre)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Catálogo de productos que ofrece cada proveedor.
-- id_producto_link es opcional: permite vincular el producto del proveedor
-- a tu catálogo propio (tabla `productos`) cuando coinciden.
CREATE TABLE proveedor_productos (
    id                  INT AUTO_INCREMENT PRIMARY KEY,
    id_proveedor        INT          NOT NULL,
    id_producto_link    INT          DEFAULT NULL,
    producto_nombre     VARCHAR(150) NOT NULL,
    variedad            VARCHAR(100) DEFAULT NULL,
    calibre             VARCHAR(50)  DEFAULT NULL,
    unidad              VARCHAR(20)  DEFAULT NULL,
    formato             VARCHAR(50)  DEFAULT NULL,
    activo              TINYINT(1)   DEFAULT 1,
    fecha_creacion      TIMESTAMP    DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_pp_prov (id_proveedor),
    INDEX idx_pp_link (id_producto_link),
    INDEX idx_pp_nombre (producto_nombre),
    CONSTRAINT fk_pp_proveedor
        FOREIGN KEY (id_proveedor) REFERENCES proveedores(id_proveedor)
        ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Historial de precios cotizados. Cada precio guardado deja registro
-- (no se sobreescribe). El "precio actual" es simplemente el último.
CREATE TABLE proveedor_cotizaciones (
    id                      INT AUTO_INCREMENT PRIMARY KEY,
    id_proveedor_producto   INT            NOT NULL,
    precio                  DECIMAL(10,2)  NOT NULL,
    validez                 ENUM('diaria','semanal','mensual') NOT NULL DEFAULT 'semanal',
    valido_hasta            DATE           NOT NULL,
    notas                   TEXT           DEFAULT NULL,
    registrado_por          VARCHAR(100)   DEFAULT NULL,
    fecha_cotizacion        TIMESTAMP      DEFAULT CURRENT_TIMESTAMP,
    INDEX idx_pc_pp (id_proveedor_producto),
    INDEX idx_pc_vence (valido_hasta),
    CONSTRAINT fk_pc_pp
        FOREIGN KEY (id_proveedor_producto) REFERENCES proveedor_productos(id)
        ON DELETE CASCADE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- Columna `direccion` agregada después: permite guardar la dirección del proveedor.
ALTER TABLE proveedores
    ADD COLUMN direccion VARCHAR(255) DEFAULT NULL AFTER ciudad;

-- Columna `foto_url` para guardar la foto del producto cotizado en terreno.
-- Cada cotización puede tener su foto propia (mostrar calidad al momento del precio).
ALTER TABLE proveedor_cotizaciones
